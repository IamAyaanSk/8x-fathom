useState<RecommendationState>("idle");

const timer = setTimeout(() => setRecommendationState("idle"), 3000);
return () => clearTimeout(timer);
}, [recommendationState]);

const {
error,
messages,
regenerate,
sendMessage,
status,
stop,
addToolApprovalResponse,
} = useChat<PortfolioAssistantUIMessage>({
transport: new DefaultChatTransport({ api: "/api/assistant" }),
sendAutomaticallyWhen: lastAssistantMessageIsCompleteWithApprovalResponses,
onFinish: async ({ messages }) => {
const userMessage = messages.findLast((m) => m.role === "user");
const query = userMessage ? getMessageText(userMessage) : "";
if (!query.trim()) return;

      setRecommendationState("loading");
      try {
        const res = await fetch("/api/assistant/recommendations", {
          method: "POST",
          headers: { "Content-Type": "application/json" },
          body: JSON.stringify({ query }),
        });

        if (!res.ok) throw new Error("Failed to fetch recommendations");

        const data = await res.json();
        const items: Recommendation[] = data?.recommendations ?? [];

        if (items.length > 0) {
          setRecommendations(items);
          setRecommendationState("loaded");
        } else {
          setRecommendations([]);
          setRecommendationState("empty");
        }
      } catch {
        console.error("Failed to fetch recommendations");
        setRecommendations([]);
        setRecommendationState("idle");
      }
    },

});

const isFetching = recommendationState === "loading";
const busy = status === "submitted" || status === "streaming" || isFetching;

let activity: string | null = null;

if (error) {
activity = "Something went wrong processing the request!";
} else if (status === "submitted") {
activity = "Thinking…";
} else if (status === "streaming") {
activity = "Crafting response…";
}

const message = input.trim();
if (!message || message.length > MAX_MESSAGE_LENGTH || busy) return;
setRecommendations([]);
setRecommendationState("idle");
sendMessage({ text: message });
setInput("");
}

                    <MessageScrollerItem messageId="assistant-recommendations">
                      <RecommendationCards recommendations={recommendations} />
                    </MessageScrollerItem>
                  )}
              </MessageScrollerContent>
            </MessageScrollerViewport>
            <MessageScrollerButton />
          </MessageScroller>
        </MessageScrollerProvider>

        <div className="p-4 pt-1">
          <form
            onSubmit={(event) => {
              event.preventDefault();
              submit();
            }}
            className="flex flex-col rounded-[24px] border border-white/10 bg-[#212121] p-3 shadow-inner transition-colors focus-within:border-white/20"
          >
            <Textarea
              value={input}
              onChange={(event) =>
                setInput(event.target.value.slice(0, MAX_MESSAGE_LENGTH))
              }
              onKeyDown={(event) => {
                if (event.key === "Enter" && !event.shiftKey) {
                  event.preventDefault();
                  submit();
                }
              }}
              maxLength={MAX_MESSAGE_LENGTH}
              disabled={busy}
              placeholder="Ask about Ayaan's work…"
              aria-label="Message the portfolio assistant"
              className="min-h-16 resize-none border-0 bg-transparent px-2 py-1 text-base md:text-sm text-foreground shadow-none placeholder:text-muted-foreground focus-visible:ring-0 disabled:bg-transparent disabled:opacity-60"
            />
            <div className="flex items-center justify-between pl-2 pr-0.5 pt-1">
              <span
                className={cn(
                  "text-[11px] transition-opacity",
                  input.length === 0 ? "opacity-0" : "opacity-100",
                  input.length === MAX_MESSAGE_LENGTH
                    ? "text-destructive"
                    : "text-muted-foreground/60",
                )}
              >
                {input.length}/{MAX_MESSAGE_LENGTH}
              </span>
              {busy ? (
                <Button
                  type="button"
                  size="icon"
                  onClick={stop}
                  className="size-8.5 rounded-full bg-secondary text-secondary-foreground hover:bg-secondary/85"
                  aria-label="Stop generation"
                >
                  <Square className="size-3 fill-current" aria-hidden="true" />
                </Button>
              ) : (
                <Button
                  type="submit"
                  size="icon"
                  disabled={!input.trim()}
                  className="size-8.5 rounded-full bg-blue-600 text-primary-foreground transition-all hover:bg-blue-500 disabled:opacity-30 disabled:pointer-events-none"
                  aria-label="Send message"
                >
                  <ArrowUp
                    className="size-4.5 stroke-[2.5]"
                    aria-hidden="true"
                  />
                </Button>
              )}
            </div>
          </form>
        </div>
      </SheetContent>
    </Sheet>

);
}
